```octave
g = gradM ( w ) ;                      # 用初始 w 设置梯度
M = findM ( w ) ;                      # 同时设置目标函数值

for l = 1:L                            # 循环 L 次
   p = randn ( size(w) ) ;             # 初始动量服从 Normal(0,1)
   H = p' * p / 2 + M ;                # 计算 H(w,p)

   p = p - epsilon * g / 2 ;           # * p 走半步
   wnew = w + epsilon * p ;            # * w 走一步
   gnew = gradM ( wnew ) ;             # * 计算新的梯度
   p = p - epsilon * gnew / 2 ;        # * p 再走半步

   Mnew = findM ( wnew ) ;             # 计算新的目标函数值
   Hnew = p' * p / 2 + Mnew ;          # 计算新的 H 值
   dH = Hnew - H ;                     # 决定是否接受
   if ( dH < 0 )                 accept = 1 ;
   elseif ( rand() < exp(-dH) ) accept = 1 ;  # 与均匀随机变量比较
   else                          accept = 0 ;
   endif
   if ( accept )    g = gnew ;   w = wnew ;   M = Mnew ;   endif
endfor

function gM = gradM ( w )              # 目标函数的梯度
   a = x * w ;                         # 计算激活量
   y = sigmoid(a) ;                    # 计算输出
   e = t - y ;                         # 计算误差
   g = - x' * e ;                      # 计算 G(w) 的梯度
   gM = alpha * w + g ;
endfunction

function M = findM ( w )               # 目标函数
   G = - (t' * log(y) + (1-t') * log( 1-y )) ;
   EW = w' * w / 2 ;
   M = G + alpha * EW ;
endfunction
```

算法 41.4　Langevin 蒙特卡罗方法的 Octave 源代码。要得到哈密顿蒙特卡罗方法，就把标有 `*` 的四行多重复几次（算法 41.8）。
